        async function initAudio() {

            try {

                await Tone.start();

                synth = new Tone.PolySynth(Tone.Synth, {

                audioReady = true;

            }

        }


        function playSound(type) {

            if (!audioReady || !synth) return;

            const now = Tone.now();

            try {

                if (type === 'move') {

                    synth.triggerAttackRelease("C5", "32n", now);

                } else if (type === 'rotate') {

                    synth.triggerAttackRelease(["E5", "G5"], "32n", now);

                } else if (type === 'drop') {

                    synth.triggerAttackRelease("G3", "16n", now);

                } else if (type === 'clear') {

                    // Cheerful "Baby Shark" style arpeggio fanfare!

                    synth.triggerAttackRelease("D5", "16n", now);

                    synth.triggerAttackRelease("E5", "16n", now + 0.12);

                    synth.triggerAttackRelease("G5", "8n", now + 0.24);

                    synth.triggerAttackRelease("G5", "8n", now + 0.40);

                    synth.triggerAttackRelease(["G5", "B5", "D6"], "4n", now + 0.56);

                } else if (type === 'gameover') {

                    synth.triggerAttackRelease("G4", "8n", now);

                    synth.triggerAttackRelease("E4", "8n", now + 0.18);

                    synth.triggerAttackRelease("C4", "4n", now + 0.36);

            } catch (e) {}

        }

            const matrix = SHAPES[shapeIdx].map(row => [...row]);

        // Collision Detection

        function checkCollision(grid, piece, offsetX = 0, offsetY = 0, testMatrix = piece.matrix) {

            for (let r = 0; r < testMatrix.length; r++) {

                for (let c = 0; c < testMatrix[r].length; c++) {

                    if (!testMatrix[r][c]) continue;

                    const newX = piece.x + c + offsetX;

                    const newY = piece.y + r + offsetY;


                    if (newX < 0 || newX >= COLS || newY >= ROWS) {

                        return true;

                    }

                    if (newY >= 0 && grid[newY][newX] !== null) {

                        return true;

                    }

                }

            }

            return false;

        }


        // Rotate Matrix 90 Degrees Clockwise

        function rotateMatrix(matrix) {

            const rows = matrix.length;

            const cols = matrix[0].length;

            const rotated = Array.from({ length: cols }, () => Array(rows).fill(0));

            for (let r = 0; r < rows; r++) {

                for (let c = 0; c < cols; c++) {

                    rotated[c][rows - 1 - r] = matrix[r][c];

                }

            }

            return rotated;

        }


        // --- 5. DRAWING EACH SHARK BLOCK (WITH EYES AND FINS!) ---

        function drawSharkBlock(ctx, gridX, gridY, cellSize, colorKey, isGhost = false) {

            ctx.fill();

            ctx.restore();

                    particles.push({

                        x: px,

                        y: centerY,

                        vx: Math.cos(angle) * speed,

                        vy: Math.sin(angle) * speed - 2.5,

                        size: Math.random() * 8 + 4,

                        color: particleColors[Math.floor(Math.random() * particleColors.length)],

                        alpha: 1,

                        isStar: Math.random() > 0.45,

                        rotation: Math.random() * Math.PI

                    });

                }

            });

        }


        function updateAndDrawParticles(ctx, cellSize) {

            // Draw row flash wave

            if (flashAlpha > 0 && flashRows.length > 0) {

                ctx.save();

                ctx.fillStyle = `rgba(255, 250, 180, ${flashAlpha})`;

                flashRows.forEach(r => {

                    ctx.fillRect(0, r * cellSize, COLS * cellSize, cellSize);

                });

                ctx.restore();

                flashAlpha -= 0.05;

            }


            // Draw explosive celebration particles

            for (let i = particles.length - 1; i >= 0; i--) {

                const p = particles[i];

                p.x += p.vx;

                p.y += p.vy;

                p.vy += 0.15; // gentle underwater gravity

                p.alpha -= 0.018;

                p.rotation += 0.08;


                if (p.alpha <= 0) {

                    particles.splice(i, 1);

                    continue;

                }


                ctx.save();

                ctx.globalAlpha = p.alpha;

                ctx.translate(p.x, p.y);

                ctx.rotate(p.rotation);

                ctx.fillStyle = p.color;


                if (p.isStar) {

                    // Draw 5-point star

                    ctx.beginPath();

                    for (let s = 0; s < 5; s++) {

                        ctx.lineTo(Math.cos((18 + s * 72) * Math.PI / 180) * p.size,

                                   -Math.sin((18 + s * 72) * Math.PI / 180) * p.size);

                        ctx.lineTo(Math.cos((54 + s * 72) * Math.PI / 180) * (p.size * 0.45),

                                   -Math.sin((54 + s * 72) * Math.PI / 180) * (p.size * 0.45));

                    }

                    ctx.closePath();

                    ctx.fill();

                } else {

                    // Draw glowing bubble

                    ctx.beginPath();

                    ctx.arc(0, 0, p.size * 0.7, 0, Math.PI * 2);

                    ctx.fill();

                }

                ctx.restore();

            }

        }


        // --- 7. GAME MECHANICS ---

        function lockPiece() {

            const cellSize = canvas.clientWidth / COLS;

            for (let r = 0; r < currentPiece.matrix.length; r++) {

                for (let c = 0; c < currentPiece.matrix[r].length; c++) {

                    if (currentPiece.matrix[r][c]) {

                        const boardY = currentPiece.y + r;

                        const boardX = currentPiece.x + c;

                        if (boardY < 0) {

                            triggerGameOver();

                            return;

                        }

                        board[boardY][boardX] = currentPiece.colorKey;

                    }

                }

            }

            playSound('drop');


            // Check for completed rows

            let clearedIndices = [];

            for (let r = ROWS - 1; r >= 0; r--) {

                if (board[r].every(cell => cell !== null)) {

                    clearedIndices.push(r);

                }

            }


            if (clearedIndices.length > 0) {

                // Trigger dramatic celebration!

                triggerDramaticRowClear(clearedIndices, cellSize);


                // Remove cleared rows and add empty rows at top

                clearedIndices.sort((a, b) => a - b);

                clearedIndices.forEach(rowIndex => {

                    board.splice(rowIndex, 1);

                    board.unshift(Array(COLS).fill(null));

                });


                // Update Score (100 per row, bonus for multi-row)

                const pointsEarned = clearedIndices.length * 100 * clearedIndices.length;

                score += pointsEarned;

                document.getElementById('score-display').textContent = score;

            }


            spawnPiece();

        }


        function moveLeft() {

            if (!isGameRunning || isPaused || !currentPiece) return;

            if (!checkCollision(board, currentPiece, -1, 0)) {

                currentPiece.x--;

                playSound('move');

            }

        }


        function moveRight() {

            if (!isGameRunning || isPaused || !currentPiece) return;

            if (!checkCollision(board, currentPiece, 1, 0)) {

                currentPiece.x++;

                playSound('move');

            }

        }


        function rotatePiece() {

            if (!isGameRunning || isPaused || !currentPiece) return;

            const rotated = rotateMatrix(currentPiece.matrix);

            // Wall kick support so toddlers never get stuck near edges

            const kicks = [0, -1, 1, -2, 2];

            for (let kick of kicks) {

                if (!checkCollision(board, currentPiece, kick, 0, rotated)) {

                    currentPiece.x += kick;

                    currentPiece.matrix = rotated;

                    playSound('rotate');

                    return;

                }

            }

        }


        function softDrop() {

            if (!isGameRunning || isPaused || !currentPiece) return;

            if (!checkCollision(board, currentPiece, 0, 1)) {

                currentPiece.y++;

                dropCounter = 0;

            } else {

                lockPiece();

            }

        }


        // Calculate Ghost Piece Y position (visual landing guide for toddlers!)

        function getGhostY() {

            let ghostY = currentPiece.y;

            while (!checkCollision(board, currentPiece, 0, ghostY - currentPiece.y + 1)) {

                ghostY++;

            }

            return ghostY;

        }


        // --- 8. MAIN RENDER LOOP ---

        function draw() {

            const width = canvas.clientWidth;

            const height = canvas.clientHeight;

            const cellSize = width / COLS;


            ctx.clearRect(0, 0, width, height);


            // Subtle underwater grid lines to help toddlers align blocks

            ctx.strokeStyle = 'rgba(255, 255, 255, 0.14)';

            ctx.lineWidth = 1;

            for (let c = 1; c < COLS; c++) {

                ctx.beginPath();

                ctx.moveTo(c * cellSize, 0);

                ctx.lineTo(c * cellSize, height);

                ctx.stroke();

            }

            for (let r = 1; r < ROWS; r++) {

                ctx.beginPath();

                ctx.moveTo(0, r * cellSize);

                ctx.lineTo(width, r * cellSize);

                ctx.stroke();

            }


            // Draw locked shark blocks on the board

            for (let r = 0; r < ROWS; r++) {

                for (let c = 0; c < COLS; c++) {

                    if (board[r][c]) {

                        drawSharkBlock(ctx, c, r, cellSize, board[r][c], false);

                    }

                }

            }


            // Draw Ghost Piece & Active Falling Shark Piece

            if (currentPiece) {

                const ghostY = getGhostY();

                for (let r = 0; r < currentPiece.matrix.length; r++) {

                    for (let c = 0; c < currentPiece.matrix[r].length; c++) {

                        if (currentPiece.matrix[r][c]) {

                            // Draw helpful ghost shadow

                            drawSharkBlock(ctx, currentPiece.x + c, ghostY + r, cellSize, currentPiece.colorKey, true);

                            // Draw active shark block

                            drawSharkBlock(ctx, currentPiece.x + c, currentPiece.y + r, cellSize, currentPiece.colorKey, false);

                        }

                    }

                }

            }


            // Draw Row Clear Dramatic Particles

            updateAndDrawParticles(ctx, cellSize);

        }


        function gameLoop(time = 0) {

            if (!isGameRunning) return;


            const deltaTime = time - lastTime;

            lastTime = time;


            if (!isPaused) {

                dropCounter += deltaTime;

                if (dropCounter > baseDropInterval) {

                    softDrop();

                    dropCounter = 0;

                }

            }


            draw();

            animationFrameId = requestAnimationFrame(gameLoop);

        }


        // --- 9. GAME STATE CONTROLS (START, RESTART, HOME, GAME OVER) ---

        function startGame() {

            initAudio();

            document.getElementById('home-screen').classList.add('hidden');

            document.getElementById('gameover-screen').classList.add('hidden');


            resizeCanvas();

            board = createEmptyBoard();

            score = 0;

            particles = [];

            flashRows = [];

            document.getElementById('score-display').textContent = '0';


            isGameRunning = true;

            isPaused = false;

            dropCounter = 0;

            lastTime = performance.now();


            spawnPiece();

            if (animationFrameId) cancelAnimationFrame(animationFrameId);

            animationFrameId = requestAnimationFrame(gameLoop);

        }


        function goHome() {

            isGameRunning = false;

            if (animationFrameId) cancelAnimationFrame(animationFrameId);

            document.getElementById('gameover-screen').classList.add('hidden');

            document.getElementById('home-screen').classList.remove('hidden');

        }


        function triggerGameOver() {

            isGameRunning = false;

            playSound('gameover');

            document.getElementById('final-score').textContent = score;

            document.getElementById('gameover-screen').classList.remove('hidden');

        }


        // --- 10. TOUCH, BUTTON & SWIPE EVENT LISTENERS (iOS & Android Friendly) ---

        function bindHoldButton(btnId, actionFn, repeatDelay = 130) {

            const btn = document.getElementById(btnId);

            let holdInterval = null;

            let holdTimeout = null;


            const startAction = (e) => {

                e.preventDefault();

                initAudio();

                btn.classList.add('pressed');

                actionFn();

                clearTimeout(holdTimeout);

                clearInterval(holdInterval);

                holdTimeout = setTimeout(() => {

                    holdInterval = setInterval(actionFn, repeatDelay);

                }, 220);

            };


            const stopAction = (e) => {

                if (e) e.preventDefault();

                btn.classList.remove('pressed');

                clearTimeout(holdTimeout);

                clearInterval(holdInterval);

            };


            btn.addEventListener('touchstart', startAction, { passive: false });

            btn.addEventListener('touchend', stopAction);

            btn.addEventListener('touchcancel', stopAction);

            btn.addEventListener('mousedown', startAction);

            btn.addEventListener('mouseup', stopAction);

            btn.addEventListener('mouseleave', stopAction);

        }


        // Bind Big Bottom Buttons

        bindHoldButton('ctrl-left', moveLeft, 120);

        bindHoldButton('ctrl-right', moveRight, 120);

        bindHoldButton('ctrl-fast', softDrop, 65);


        // Rotate button (single tap per spin so toddlers don't accidentally over-spin)

        const btnRotate = document.getElementById('ctrl-rotate');

        const handleRotate = (e) => {

            e.preventDefault();

            initAudio();

            btnRotate.classList.add('pressed');

            rotatePiece();

            setTimeout(() => btnRotate.classList.remove('pressed'), 120);

        };

        btnRotate.addEventListener('touchstart', handleRotate, { passive: false });

        btnRotate.addEventListener('mousedown', handleRotate);


        // Navigation & Screen Buttons

        document.getElementById('btn-play').addEventListener('click', startGame);

        document.getElementById('btn-play-again').addEventListener('click', startGame);

        document.getElementById('btn-restart').addEventListener('click', startGame);

        document.getElementById('btn-home').addEventListener('click', goHome);


        // Speed Mode Toggle (Toddler vs Big Kid)

        const modeToddler = document.getElementById('mode-toddler');

        const modeKid = document.getElementById('mode-kid');

        modeToddler.addEventListener('click', () => {

            baseDropInterval = 1150;

            modeToddler.classList.add('active');

            modeKid.classList.remove('active');

        });

        modeKid.addEventListener('click', () => {

            baseDropInterval = 700;

            modeKid.classList.add('active');

            modeToddler.classList.remove('active');

        });


        // Direct Canvas Touch Gestures (Tap canvas to Spin, Swipe Left/Right/Down to move)

        let touchStartX = 0;

        let touchStartY = 0;

        let touchMoved = false;


        canvas.addEventListener('touchstart', (e) => {

            e.preventDefault();

            initAudio();

            const touch = e.touches[0];

            touchStartX = touch.clientX;

            touchStartY = touch.clientY;

            touchMoved = false;

        }, { passive: false });


        canvas.addEventListener('touchmove', (e) => {

            e.preventDefault();

            if (!isGameRunning) return;

            const touch = e.touches[0];

            const dx = touch.clientX - touchStartX;

            const dy = touch.clientY - touchStartY;

            const stepThreshold = 28;


            if (Math.abs(dx) > stepThreshold) {

                if (dx > 0) moveRight();

                else moveLeft();

                touchStartX = touch.clientX;

                touchMoved = true;

            }

            if (dy > stepThreshold) {

                softDrop();

                touchStartY = touch.clientY;

                touchMoved = true;

            }

        }, { passive: false });


        canvas.addEventListener('touchend', (e) => {

            e.preventDefault();

            if (!touchMoved && isGameRunning) {

                rotatePiece(); // Tap on board rotates shark block

            }

        }, { passive: false });


        // Keyboard Support (for iPad with keyboard or desktop testing)

        window.addEventListener('keydown', (e) => {

            if (!isGameRunning) return;

            if (e.key === 'ArrowLeft') moveLeft();

            else if (e.key === 'ArrowRight') moveRight();

            else if (e.key === 'ArrowUp' || e.key === ' ') rotatePiece();

            else if (e.key === 'ArrowDown') softDrop();

        });


        // Initial setup on window load

        window.onload = () => {

            resizeCanvas();

        };

    </script>
